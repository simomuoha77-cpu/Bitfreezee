 API Documentation

          

            This calls JuanAi's real backend server (server.js) over plain HTTP — works from any server, including BetaKE's, since it's a normal network request instead of browser-only localStorage.
          

          

// From BetaKE's Node.js backend (or any server):

const API_KEY = 'jsk_xxxxxxxxxxxxxxxx';

const JUANAI_URL = 'https://your-juanai-domain.com'; // or http://localhost:3000 for local dev


async function getJuanAiFixtures(days) {

  const resp = await fetch(`${JUANAI_URL}/api/fixtures?key=${API_KEY}&days=${days}`);

  if (!resp.ok) throw new Error('JuanAi API error: ' + resp.status);

  const data = await resp.json();

  return data.matches || [];

}


// Each match has:

// match.homeTeam, match.awayTeam

// match.utcDate, match.status

// match.minute — current match minute if IN_PLAY (null otherwise)

// match.minuteIsEstimated — true if match.minute is an ESTIMATE (odds-api.io

//   sourced matches don't have a real match clock, only kickoff time — see

//   note below). false/absent means it's a real minute from football-data.org.

//   Show a "~" prefix or similar when true, so users know it's approximate.

// match.competition

// match.aiOdds.homeWin, match.aiOdds.draw, match.aiOdds.awayWin

// match.aiOdds.over25, match.aiOdds.under25

// match.aiOdds.btts, match.aiOdds.bttsNo

// match.aiPrediction, match.aiConfidence

// match.aiAnalysis, match.aiXG, match.aiValueBet

// match.aiOdds.isRealMarketOdds — true = real bookmaker price (safe for

//   real-money bets), false/absent = AI estimate (NOT safe for real-money

//   bets). Pass &realOddsOnly=1 on the fetch URL to have this filtered

//   server-side automatically.


// ⚠️ match.aiOdds are AI-generated predictions, not real bookmaker odds,

// UNLESS aiOdds.isRealMarketOdds is true. Label accordingly if shown to end users.

// In BetaKE's Express route, e.g. GET /games/today:

app.get('/games/today', async (req, res) => {

  try {

    const games = await getJuanAiFixtures(0);

    res.json({ games });

  } catch (e) {

    res.status(502).json({ error: 'Could not reach JuanAi' });

  }

});
